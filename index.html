/** * RÔLE : Point d'entrée de l'application. Initialise le chargement après construction du DOM. */ document.addEventListener("DOMContentLoaded", () => { analyserFichierOPML(); }); /** * RÔLE : Récupère le fichier local 'feeds.opml', extrait les URL des flux et lance leur traitement. * LOGIQUE DE SÉCURITÉ : Vérifie la validité de la réponse HTTP et la conformité du code XML * afin d'éviter les blocages silencieux en cas de désynchronisation de fichiers. */ async function analyserFichierOPML() { const conteneurPrincipal = document.getElementById("flux-container"); try { const reponse = await fetch('feeds.opml'); // GESTION DES ERREURS HTTP : fetch ne levant pas d'erreur sur un 404, validation explicite requise. if (!reponse.ok) { throw new Error(`Impossible d'accéder au fichier 'feeds.opml' (Statut HTTP : ${reponse.status}).`); } const chaineDonnees = await reponse.text(); const parseur = new DOMParser(); const documentXML = parseur.parseFromString(chaineDonnees, "text/xml"); // GESTION DES ERREURS DE PARSING : Vérification de la présence d'une balise d'erreur XML. if (documentXML.querySelector("parsererror")) { throw new Error("Le fichier 'feeds.opml' n'est pas un document XML/OPML valide (Erreur de syntaxe)."); } const elementsOutline = documentXML.querySelectorAll("outline[xmlUrl]"); // Cas où le fichier est valide mais ne contient aucun flux exploitable if (elementsOutline.length === 0) { conteneurPrincipal.innerHTML = "

Aucun flux valide trouvé dans le fichier OPML.

"; return; } // Algorithme de déploiement des cartes de veille elementsOutline.forEach(outline => { const titreSource = outline.getAttribute("title") || outline.getAttribute("text") || "Source inconnue"; const urlFlux = outline.getAttribute("xmlUrl"); if (urlFlux) { creerBlocStructurel(titreSource, urlFlux); chargerArticlesFlux(titreSource, urlFlux); } }); } catch (erreur) { console.error("Échec de l'initialisation de la veille :", erreur); // INJECTION PROTOCOLAIRE D'ERREUR : Permet un diagnostic direct sur l'interface utilisateur. conteneurPrincipal.innerHTML = `

[Erreur de chargement de l'infrastructure]

${erreur.message}

Solution : Vérifiez que le fichier feeds.opml est bien nommé en minuscules et présent à la racine de votre dépôt principal.
`; } } /** * RÔLE : Instancie une section HTML vide dans le conteneur pour garantir la structure de la grille. */ function creerBlocStructurel(titreSource, urlFlux) { const conteneurPrincipal = document.getElementById("flux-container"); const section = document.createElement("section"); section.className = "feed-section"; // LOGIQUE DE NETTOYAGE : Remplacement de btoa par une Regex pour éliminer tout risque lié aux caractères spéciaux. const idValide = urlFlux.replace(/[^a-zA-Z0-9]/g, ""); section.id = `flux-${idValide}`; section.innerHTML = `

${titreSource}

Synchro en cours...
`; conteneurPrincipal.appendChild(section); } /** * RÔLE : Interroge le flux distant via un proxy CORS, extrait les articles et les insère dans sa section. */ async function chargerArticlesFlux(titreSource, urlFlux) { const idValide = urlFlux.replace(/[^a-zA-Z0-9]/g, ""); const conteneurArticles = document.querySelector(`#flux-${idValide} .articles-list`); const urlRequeteProxiee = `https://corsproxy.io/?${encodeURIComponent(urlFlux)}`; try { const reponse = await fetch(urlRequeteProxiee); if (!reponse.ok) throw new Error(`HTTP ${reponse.status}`); const texteBrut = await reponse.text(); const parseur = new DOMParser(); const xmlDoc = parseur.parseFromString(texteBrut, "text/xml"); const estAtom = xmlDoc.querySelector("entry") !== null; const balisesArticles = estAtom ? xmlDoc.querySelectorAll("entry") : xmlDoc.querySelectorAll("item"); conteneurArticles.innerHTML = ""; if (balisesArticles.length === 0) { conteneurArticles.innerHTML = "

Aucun article disponible.

"; return; } const selectionArticles = Array.from(balisesArticles).slice(0, 4); selectionArticles.forEach(article => { const titreArticle = article.querySelector("title")?.textContent || "Article sans titre"; let lienArticle = estAtom ? article.querySelector("link")?.getAttribute("href") : article.querySelector("link")?.textContent; if (!lienArticle || lienArticle === "#") return; const elementAncre = document.createElement("a"); elementAncre.className = "article-link"; elementAncre.href = lienArticle.trim(); elementAncre.target = "_blank"; elementAncre.textContent = titreArticle; conteneurArticles.appendChild(elementAncre); }); } catch (erreur) { conteneurArticles.innerHTML = "

Flux temporairement indisponible.

"; } }
